```sql
select 
	PSN.id AS P_ID,
	PSN.NAME AS P_NAME,
	PSN.address_number AS ADDRESS_NO,
	PSN.address_street_name AS ADDRESS_NAME,
	PSN.ssn AS P_SSN,
	DL.id AS DRIVING_LICENCE_ID,
	DL.age AS AGE,
	DL.height AS HEIGHT,
	DL.hair_color AS HAIR_COLOR,
	DL.gender AS GENDER,
	DL.plate_number AS CAR_PLATE_NO,
	DL.car_make AS CAR_BRAND,
	DL.car_model AS CAR_MODEL
FROM 
	person AS PSN
	
	JOIN drivers_license AS DL
	ON PSN.license_id = DL.id
```
