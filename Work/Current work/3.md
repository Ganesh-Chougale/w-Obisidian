        public decimal GetPrdListForPurchace(int SubGlAcNo, int materialcode, string month, Int64 trnno, int AmendNo)
        {
            try
            {
                decimal ret = 0;
                string query = "SELECT ISNULL(SUM(TEMP.PRD_PLAN_QTY),0) - ISNULL(TRN_SCH.SCH_QTY,0) AS PRD_PLAN_QTY";
                query = query + " FROM(";
                query = query + " SELECT ISNULL(SUM(PLAN_QTY), 0) PRD_PLAN_QTY FROM TRN_PRD_CONSUMABLE_OUT_SOURCE  WHERE SUB_GL_ACNO = " + SubGlAcNo + " AND MATERIAL_CODE = " + materialcode + " and LEFT(TRN_DATE, 6) = LEFT(" + month + ", 6) AND STATUS_CODE = 0";
                query = query + " UNION ALL";
                query = query + " SELECT DISTINCT  CASTING_NORMS AS PRD_PLAN_QTY  FROM TRN_PRD_CONSUMABLE_OUT_SOURCE  WHERE SUB_GL_ACNO = " + SubGlAcNo + " AND MATERIAL_CODE = " + materialcode + " and LEFT(TRN_DATE, 6) = LEFT(" + month + ", 6) AND STATUS_CODE = 0";
                query = query + " )TEMP";
                query = query + " LEFT OUTER JOIN(";
                query = query + " SELECT";
                query = query + " 	SUM(SCH_QTY) SCH_QTY";
                query = query + " FROM ";
                query = query + " 	TRN_SCHEDULE_I ";
                query = query + " 	INNER JOIN TRN_SCHEDULE_H ON TRN_SCHEDULE_H.TRN_NO = TRN_SCHEDULE_I.TRN_NO";
                query = query + " 		AND TRN_SCHEDULE_H.AMEND_NO = TRN_SCHEDULE_I.AMEND_NO";
                query = query + " WHERE ";
                query = query + " 	SUBSTRING(CONVERT(VARCHAR,TRN_SCHEDULE_H.TRN_NO),8,3)=903";
                query = query + " 	AND TRN_SCHEDULE_H.TRN_SUB_TYPE IN(14,17)";
                query = query + " 	AND TRN_SCHEDULE_H.SUB_GL_ACNO = " + SubGlAcNo;
                query = query + " 	AND LEFT(TRN_SCHEDULE_I.SCH_DATE,6) = LEFT(" + month + ",6)";
                query = query + " 	AND TRN_SCHEDULE_I.MATERIAL_CODE = " + materialcode;
                query = query + " 	AND TRN_SCHEDULE_I.TRN_NO != " + trnno;
                query = query + " 	AND TRN_SCHEDULE_I.AMEND_NO != " + AmendNo;
                query = query + " 	AND TRN_SCHEDULE_I.STATUS_CODE IN(0,101)";
                query = query + " )TRN_SCH ON 1=1";
                query = query + " GROUP BY ISNULL(TRN_SCH.SCH_QTY,0)";
                using (ConManager con = new ConManager())
                {
                    ret = (decimal)con.transactionDb.ExecuteScalar(query);
                }
                return ret;
            }
            catch (Exception ex)
            {
                throw ex;
            }
        }
