    $("#save").click(function () {

        debugger;
        var currentDate = new Date().toISOString().slice(0, 10).replace(/-/g, "");
        var CustomerCodeVal = $("#hfCustomerCode").val();
        var SellerCompanyIdVal = $("#hfSellerCompanyId").val();
        var PackingDestinationVal = $("#ddlPackingDestination").val();
        var EffectiveDateVal = $("#dtEffectiveDate").val();
        var RemarkVal = $("#txtRemark").val();

        // Checks
        if (EffectiveDateVal == "" || EffectiveDateVal == undefined || EffectiveDateVal == null) {
            customWarning("#", "Effective Date", "please select effective date", "#C46A69", 3000);
            return false;
        }



        // Part Codes

        var iList = [];
        $("#PartQuantityTable tbody tr").each(function () {

            var PartCodeVal = $(this).find(".hfPartCode").val();
            var PerBoxPartQuantityVal = $(this).find(".partQuantityVal").val();

            if (PerBoxPartQuantityVal == "" ||
                PerBoxPartQuantityVal == undefined ||
                PerBoxPartQuantityVal == null) {

                customWarning("#", "Per Box Part Quantity", "please input quantity properly", "#C46A69", 3000);

                return false;
            }

            iList.push({
                MaterialCode: PartCodeVal,
                PerBoxPartQty: PerBoxPartQuantityVal
            });

        });


        if ($("#PackingMaterialBomDetails tbody tr").length <= 0) {
            customWarning("#", "No Packing Material", "Please add at least one packing material in the BOM DETAILS table.", "#C46A69", 3000);
            return false;
        }

        var iBomList = [];
        $("#PackingMaterialBomDetails tbody tr").each(function () {

            var MaterialCodeVal = $(this).find(".tdhfMatCode").val();
            var QuantityVal = $(this).find(".tdhfQty").val();
            var BomTypeVal = $(this).find(".bomType").val();
            var MaterialRateVal = $(this).find(".tdhfmaterialRate").val();

            if (QuantityVal == "" ||
                QuantityVal == undefined ||
                QuantityVal == null) {

                customWarning("#", "Per Quantity", "please input quantity properly", "#C46A69", 3000);
                supplierValidationFailed = true;
                return false;
            }

            iBomList.push({
                MaterialCode: MaterialCodeVal,
                Quantity: QuantityVal,
                BomType: BomTypeVal,
                Rate: MaterialRateVal
            });

        });

        var iBomSupList = [];
        var supplierValidationFailed = false;

        $("#PackingMaterialBomDetails tbody tr").each(function () {

            var currentRow = $(this);
            var materialCode = currentRow.find(".tdhfMatCode").val();
            var materialName = currentRow.find(".tdhfMatName").val();
            var supplierData = currentRow.find(".supplierData").val();
            var suppliers = [];

            try {
                suppliers = supplierData ? JSON.parse(supplierData) : [];
            }
            catch (e) {
                suppliers = [];
            }


            if (!Array.isArray(suppliers) || suppliers.length === 0) {

                customWarning("#", "Supplier Required", "Please add at least one supplier for material: <br> <b>" + materialName + "</b>.", "#C46A69", 4000);
                supplierValidationFailed = true;
                return false;
            }

            $.each(suppliers, function (index, supplier) {

                iBomSupList.push({
                    MaterialCode: materialCode,
                    SubGlAcNo: supplier.SupplierCode,
                    SupplierPercentage: supplier.Percentage
                });

            });

        });

        if (supplierValidationFailed) {
            return false;
        }


        var ParaObj = {
            TrnDate: currentDate,
            CustomerCode: CustomerCodeVal,
            SellerCompanyCode: SellerCompanyIdVal,
            PackingDestination: PackingDestinationVal,
            EffectiveDate: EffectiveDateVal,
            Remark: RemarkVal,
            LstPackingDetailsI: iList,
            LstPackingDetailsIBom: iBomList,
            LstPackingDetailsIBomSup: iBomSupList
        }

        console.log(ParaObj);
        //return false;

        $.ajax({
            url: 'PackingDetails/CreateSave',
            type: "POST",
            async: false,
            contentType: "application/json",
            dataType: "json",
            data: JSON.stringify(ParaObj),
            success: function (response, status, xhr) {
                if (response === 0) {
                    var urlName = 'PackingDetails/index';
                    $.ajax({
                        url: urlName,
                        type: "GET",
                        success: function (response, status, xhr) {
                            $("#mainContent").html(response);
                        },
                        error: function (err) {
                            $("#mainContent").html(err);
                        }
                    });
                }
            },
            error: function (xhr) {
                alert("Error: " + xhr.status + ": " + xhr.statusText);
            }
        });
    });
