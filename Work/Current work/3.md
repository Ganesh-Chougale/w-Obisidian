    function searchCustomer() {
        var selectedCustomers = $('#ddlCustomer').val();

        if (selectedCustomers != null) {
            var sendCustomers = selectedCustomers.join(',');
            debugger;
            var urlName = "PackingCost/CustomerIdWiseList?CustomerIds=" + sendCustomers;
            $.ajax({
                url: urlName,
                dataType: "json",
                type: "POST",
                contentType: "application/json; charset=utf-8",
                success: function (data) {
                    debugger;

                    if (data.length > 0) {
                        var distinctCustomerCount = new Set(data.map(x => x.SubGlAcno)).size;
                        var distinctMaterialCount = new Set(data.map(x => x.MaterialCode)).size;
                        $('#lblTotalCustomer').html(distinctCustomerCount);
                        $('#lblTotalMaterial').html(distinctMaterialCount);

                        //var greenCount = 0;
                        //var redCount = 0;

                        var greenPackingCount = 0;
                        var greenChildPartCount = 0;
                        var greenTransportCount = 0;

                        var redPackingCount = 0;
                        var redChildPartCount = 0;
                        var redTransportPartCount = 0;

                        $("#tblCustomerWiseMaterialList > tbody").empty();
                        $.each(data, function (i, item) {
                            
                            (item.TargetPackingAgree <= item.TargetPackingActual) ? redPackingCount++ : greenPackingCount++;

                            (item.TargetChildPartActual <= item.TargetChildPartAgree) ? redChildPartCount++ : greenChildPartCount++;

                            (item.TargetTransportActual <= item.TargetTransportAgree) ? redTransportPartCount++ : greenTransportCount++;

                            var row = `
                                        <tr>

                                            <td class ="left-push">
                                                ${item.CustomerName}
                                                <input type="hidden" class ="clsSubGlAcno" value="${item.SubGlAcno}" />
                                            </td>

                                            <td class ="left-push">
                                                ${item.MaterialName}
                                                <input type="hidden" class ="clsMaterialCode" value="${item.MaterialCode}" />
                                            </td>

                                            <td style="text-align:center;">
                                                ${item.TargetPackingAgree}
                                                <input type="hidden" class ="clsTargetPackingAgree" value="${item.TargetPackingAgree}" />
                                            </td>

                                            <td style="text-align:center;">
                                                ${item.TargetPackingPrice}
                                                <input type="hidden" class ="clsTargetPackingPrice" value="${item.TargetPackingPrice}" />
                                            </td>

                                            <td style="text-align:center;">
                                                ${item.TargetPackingActual}
                                                <input type="hidden" class ="clsTargetPackingActual" value="${item.TargetPackingActual}" />
                                            </td>

                                            <td style="text-align:center;">
                                                ${item.TargetChildPartAgree}
                                                <input type="hidden" class ="clsTargetChildPartAgree" value="${item.TargetChildPartAgree}" />
                                            </td>

                                            </td>

                                            </td>

                                        </tr>
                                        `;
                            $("#tblCustomerWiseMaterialList > tbody").append(row);
