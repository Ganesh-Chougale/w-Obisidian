```sql
SELECT
	EMP.EMPLOYEE_ID AS 'EMP ID',
	FORMAT(CAST(EMP.JOIN_DATE AS DATE), 'yyyy/dd/MMM') AS JOIN_DATE,
	CONCAT(EMP.EMPLOYEE_FIRST_NAME, '  ', EMP.EMPLOYEE_MIDDLE_NAME, '  ', EMP.EMPLOYEE_LAST_NAME) AS 'FULL NAME',
	FORMAT(CAST(EMP.BIRTH_DATE AS DATE), 'yyyy/dd/MMM') AS 'BDAY',
	DEP.PARENT_DEPARTMENT_ID AS 'DEP ID',
	DEP.DEPARTMENT_NAME AS 'Department Name',
	MH.SALARY AS 'Salary',
	ISNULL(FORMAT(CONVERT(DATETIME, EMP_HIST.TRN_DATE),  'dd/MM/yyyy'), '') AS TRN_DATE,
	ISNULL(FORMAT(CONVERT(DATETIME, EMP_HIST.TRN_DATE),  'yyyy'), '') AS YEAR_NO,
	MH.CTC AS 'CTC'

FROM
	ERP_MASTERS..MST_EMPLOYEE AS EMP

	JOIN ERP_MASTERS..MST_DEPARTMENT AS DEP
	ON EMP.PAYROLL_DEPT = DEP.DEPARTMENT_ID

	JOIN MST_EMPLOYEE_SALARY_H AS MH
	ON EMP.EMPLOYEE_ID = MH.EMPLOYEE_CODE

	JOIN MST_EMPLOYEE_SALARY_HISTORY AS EMP_HIST
	ON EMP_HIST.EMPLOYEE_CODE = MH.EMPLOYEE_CODE

WHERE
	EMP.STATUS_CODE = 0
	--AND DEP.DEPARTMENT_ID = '1010000025'

ORDER BY
	MH.SALARY DESC,
	JOIN_DATE
```   
