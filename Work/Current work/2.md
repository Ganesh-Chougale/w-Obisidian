        public async Task<ActionResult> GoCreate(string sortOrder, string sortdir, int? page)
        {
            string strSortDir = "";
            MstMenu _objMstMenu = MstMenu.getMenuByDocNo(TargetPriceAmend.MenuDocNo);
            List<TargetPriceAmend> retList = await TargetPriceAmend.GetAmendGoCreateList();
            ViewBag.CurrentSort = sortOrder;

            if (!string.IsNullOrEmpty(sortdir))
            {
                strSortDir = sortdir;
                if (sortdir == "desc")
                {
                    ViewBag.SortDir = "asc";
                }
                else
                {
                    ViewBag.SortDir = "desc";
                }
            }

            switch (sortOrder)
            {
                case "TrnNo":
                    if (strSortDir == "desc")
                    {
                        retList = retList.OrderByDescending(obj => obj.TrnNo).ToList();
                    }
                    else
                    {
                        retList = retList.OrderBy(obj => obj.TrnNo).ToList();
                    }
                    break;
                case "DateSort":
                    if (strSortDir == "desc")
                    {
                        retList = retList.OrderByDescending(obj => obj.FormattedTrnDate).ToList();
                    }
                    else
                    {
                        retList = retList.OrderBy(obj => obj.FormattedTrnDate).ToList();
                    }
                    break;
                case "SellerComp":
                    if (strSortDir == "desc")
                    {
                        retList = retList.OrderByDescending(obj => obj.SellerCompanyName).ToList();
                    }
                    else
                    {
                        retList = retList.OrderBy(obj => obj.SellerCompanyName).ToList();
                    }
                    break;
                case "Customer":
                    if (strSortDir == "desc")
                    {
                        retList = retList.OrderByDescending(obj => obj.CustomerName).ToList();
                    }
                    else
                    {
                        retList = retList.OrderBy(obj => obj.CustomerName).ToList();
                    }
                    break;
                case "CrtBy":
                    if (strSortDir == "desc")
                    {
                        retList = retList.OrderByDescending(obj => obj.CrtBy).ToList();
                    }
                    else
                    {
                        retList = retList.OrderBy(obj => obj.CrtBy).ToList();
                    }
                    break;
                default:
                    retList = retList.OrderByDescending(obj => obj.TrnNo).ToList();
                    ViewBag.SortDir = "desc";
                    break;
            }

            int pageSize = 50;
            int pageNumber = 1;
            if (page != null)
            {
                pageNumber = Convert.ToInt16(page);
            }

            return PartialView(retList.ToPagedList(pageNumber, pageSize));
        }
