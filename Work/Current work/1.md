## 3. `closePopup_Supplier()`, `handleSupAddRow()`, `saveSupplierPopup()`, `calculateSupplierTotal()`
```csharp
    function closePopup_Supplier() {
        $('#PopUpUI_Supplier').hide();
    }

    function handleSupAddRow() {

        if (!currentSupplierRow || currentSupplierRow.length === 0) {
            customWarning("#", "Supplier", "Please select a material first.", "#C46A69", 3000);
            return;
        }

        var supplierCode = $("#ddlSupplier").val();

        if (!supplierCode) {
            customWarning("#ddlSupplier", "Supplier", "Please select a supplier.", "#C46A69", 3000);
            return;
        }

        var supplierName = $("#ddlSupplier option:selected").text();

        var alreadyExists = false;

        $("#Supplier_Table tbody tr").each(function () {

            var existingCode = $(this).find(".supplierCode").val();

            if (existingCode == supplierCode) {
                alreadyExists = true;
                return false;
            }
        });

        if (alreadyExists) {

            customWarning(
                "#ddlSupplier", "Supplier", "This supplier is already added.", "#C46A69", 3000);

            return;
        }

        var rowCount = $("#Supplier_Table tbody tr").length;

        var row = `
                    <tr>
                        <td style="text-align:center;">
                            ${rowCount + 1}
                        </td>

                        <td>
                            ${supplierName}
                            <input type="hidden" class="supplierCode" value="${supplierCode}" />
                        </td>

                        <td>
                            <input type="text" class="supplierPercentage" style="width:100%; text-align:right;" oninput="validateDecimalInput(this); calculateSupplierTotal();" />
                        </td>
                    </tr>
                `;

        $("#Supplier_Table tbody").append(row);

        calculateSupplierTotal();
    }

    function saveSupplierPopup() {

        if (!currentSupplierRow || currentSupplierRow.length === 0) {
            closePopup_Supplier();
            return;
        }

        var total = calculateSupplierTotal();

        if (Math.abs(total - 100) > 0.001) {
            customWarning("#supplierPercentageTotal", "Supplier Percentage", "Total Supplier Percentage must be exactly 100%. Current total is " + total.toFixed(2) + "%.", "#C46A69", 4000);
            return;
        }

        var suppliers = [];

        $("#Supplier_Table tbody tr").each(function () {
            var supplierCode = $(this).find(".supplierCode").val();
            var supplierName = $(this).find("td:eq(1)").clone().children().remove().end().text().trim();
            var percentage = parseFloat($(this).find(".supplierPercentage").val()) || 0;

            suppliers.push({
                SupplierCode: supplierCode,
                SupplierName: supplierName,
                Percentage: percentage
            });
        });

        currentSupplierRow.find(".supplierData").val(JSON.stringify(suppliers));

        console.log(suppliers);
        customWarning("", "Suppliers Saved", "Material wise suppliers attatched", "#5B835B", 4000);
        closePopup_Supplier();
    }

    function calculateSupplierTotal() {
        var total = 0;

        $("#Supplier_Table tbody tr").each(function () {
            var percentage = parseFloat($(this).find(".supplierPercentage").val()) || 0;
            total += percentage;
        });

        $("#supplierPercentageTotal").text(total.toFixed(2));

        return total;
    }
```
