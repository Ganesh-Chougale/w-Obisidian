        public static async Task<List<PackingDetailsIBom>> GetTrnNoWiseData(Int64 TrnNo, int AmendNo)
        {
            try
            {
                ITrnPackingDetailsIBom interfaceObj = new DALTrnPackingDetailsIBom();
                List<PackingDetailsIBom> lst = new List<PackingDetailsIBom>();
                List<DTOTrnPackingDetailsIBom> dtoList = interfaceObj.GetExisting(TrnNo, AmendNo);
                List<MaterialsEntity> matList = await MaterialsService.GetMaterialFillList(dtoList.Select(x => x.MATERIAL_CODE).ToList());
                List<StockGroup> stkgrpList = await StockGroup.GetAllAsync();


                if (dtoList.Count > 0)
                {
                    lst = (
                            from obj in dtoList

                            join mat in matList
                            on obj.MATERIAL_CODE equals mat.MaterialCode

                            join slg in stkgrpList
                            on mat.StockListGroup equals slg.StockGroupCcode

                            select new PackingDetailsIBom
                            {
                                TrnNo = obj.TRN_NO,
                                SrNo = obj.SR_NO,
                                TrnDate = obj.TRN_DATE,
                                TrnSubType = obj.TRN_SUB_TYPE,
                                AmendNo = obj.AMEND_NO,
                                MaterialCode = obj.MATERIAL_CODE,
                                MaterialName = mat.MaterialName,
                                Quantity = obj.QUANTITY,
                                StatusCode = obj.STATUS_CODE,
                                DrawingNo = mat.DrawingNo,
                                StockListGroupCode = mat.StockListGroup,
                                StockListGroupName = slg.LongName,
                                MaterialUOM = mat.MaterialUom,
                                //Rate = Rate
                            }
                           ).ToList();
                }
                return lst;
            }
            catch (Exception ex)
            {
                throw ex;
            }
        }
