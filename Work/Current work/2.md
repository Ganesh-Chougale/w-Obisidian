public static async Task<List<PendingInterSoList>> GetRentalOrderList(int SoType, int CompanyId)
{
try
{
_dalPendingInterSoList = new DALPendingInterSoList();
List<PendingInterSoList> rentalList = new List<PendingInterSoList>();
List<DTOPendingInterSoList> dtoList = await Task.Run(() => _dalPendingInterSoList.GetRentalOrderList(SoType, CompanyId));

if (dtoList == null || !dtoList.Any())
{
return rentalList;
}

List<MaterialsEntity> lstCustomerMaterials = await MaterialsService.GetMaterialFillList(dtoList.Where(x => x.CUSTOMER_MATERIAL_CODE != null).Select(x => x.CUSTOMER_MATERIAL_CODE).Distinct().ToList());
List<MaterialsEntity> lstMappedMaterials = await MaterialsService.GetMaterialFillList(dtoList.Where(x => x.MAPPED_MATERIAL_CODE != null).Select(x => x.MAPPED_MATERIAL_CODE).Distinct().ToList());
List<AccountSubGlEntity> LstCustomer = await AccountSubGlService.GetSubGlFillList(dtoList.Where(x => x.CUSTOMER_CODE != null).Select(x => x.CUSTOMER_CODE).Distinct().ToList());

lstCustomerMaterials = lstCustomerMaterials ?? new List<MaterialsEntity>();
lstMappedMaterials = lstMappedMaterials ?? new List<MaterialsEntity>();
LstCustomer = LstCustomer ?? new List<AccountSubGlEntity>();

rentalList =
(from objDto in dtoList

join custMat in lstCustomerMaterials
on objDto.CUSTOMER_MATERIAL_CODE equals custMat.MaterialCode
into custMatList
from custMatObj in custMatList.DefaultIfEmpty()

join mappedMat in lstMappedMaterials
on objDto.MAPPED_MATERIAL_CODE equals mappedMat.MaterialCode
into MapMatList
from mapMatObj in MapMatList.DefaultIfEmpty()

join objCustomer in LstCustomer
on objDto.CUSTOMER_CODE equals objCustomer.SubGlAcNo

select new PendingInterSoList
{
TrnNo = objDto.TRN_NO,
ShortTrnNo = objDto.TRN_NO.ToString().Substring(3, 4) + "-" + objDto.TRN_NO.ToString().Substring(objDto.TRN_NO.ToString().Length - 6),
AmendNo = objDto.AMEND_NO,
TrnDate = objDto.TRN_DATE,
FormattedTrnDate = DateUtility.getFormatedDate(objDto.TRN_DATE, 0),
CustomerCode = objDto.CUSTOMER_CODE,
CustomerName = objCustomer != null ? objCustomer.LongName : "",
CustomerMaterialCode = objDto.CUSTOMER_MATERIAL_CODE,
CustomerMaterialName = custMatObj != null ? custMatObj.MaterialName : "",
PoQty = objDto.PO_QTY,
Rate = objDto.RATE,
MappedMaterialCode = objDto.MAPPED_MATERIAL_CODE,
MappedMaterialName = mapMatObj != null ? mapMatObj.MaterialName : ""
}).ToList();

return rentalList;
}
catch (Exception ex)
{
throw ex;
}
}

                CustStateCode = _objState.StateCode;

            return PartialView();
        }

    function calculateAmt(obj) {
        var $row = $(obj).closest('tr');

        var NQty = parseFloat($row.find("[id^=Qty]").val()) || 0;
        var NRate = parseFloat($row.find("[id^=Rate]").val()) || 0;
        var NPerRate = parseFloat($row.find("[id^=PerRate]").val()) || 0;
        var NDisc = parseFloat($row.find("[id^=Disc]").val()) || 0;
        var CastWt = parseFloat($row.find("[id^=CastingWt]").val()) || 0;
        var isPerKg = $row.find("[id^=IsPerKg]").is(":checked");

        var total = 0.00;
        var discRate = 0.00;

        if (NPerRate > 0) {
            NRate = NRate / NPerRate;
        }

        if (isPerKg) {
            NRate = NRate * CastWt;
        }

        total = NRate * NQty;

        if (NDisc !== 0) {
            discRate = NRate - parseFloat(((NRate / 100.00) * NDisc).toFixed(2));
            var disc = (total * NDisc) / 100;
            total = total - disc;
        } else {
            discRate = NRate;
        }

        $row.find("[id^=Amount]").val(total.toFixed(2));
        $row.find("[id^=DisRate]").val(discRate.toFixed(2));

        calculateSum();
    }
