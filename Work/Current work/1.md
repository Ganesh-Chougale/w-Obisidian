```csharp
--SELECT * FROM MST_BRANCH_DEPT_MIN_MAX


SELECT
	MBDMM.MATERIAL_CODE,
	SUM(MBDMM.MIN_QTY) AS SUM_OF_MIN_QTY,
	SUM(MBDMM.MAX_QTY) AS SUM_OF_MAX_QTY
FROM
	MST_BRANCH_DEPT_MIN_MAX AS MBDMM

	INNER JOIN ERP_MASTERS..MST_MATERIALS AS M_MAT
	ON MBDMM.MATERIAL_CODE = M_MAT.MATERIAL_CODE

WHERE
	MBDMM.BRANCH_CODE = 1010000003
	AND M_MAT.STOCK_LIST_GROUP = 1010000138
	AND MBDMM.STATUS_CODE IN (0, 101)

GROUP BY
	MBDMM.MATERIAL_CODE
```
