USE [SRF_ERP]

CREATE PROCEDURE [dbo].[SP_TRN_NO]        
     @TRN_NO NVARCHAR(100)
AS 


DECLARE  @T_FINAL TABLE (STATEMENT_QRY VARCHAR(MAX), TABLE_TYPE VARCHAR(50))
DECLARE  @T_TRN_NO TABLE (TRN_NO VARCHAR(MAX))
DECLARE  @T TABLE (ID int identity(1,1),TABLE_NAME VARCHAR(200), SCHEME VARCHAR(100), COLUMN_NAME VARCHAR(200))

INSERT INTO @T 

SELECT 
    t.name AS table_name,
    SCHEMA_NAME(schema_id) AS schema_name,
    c.name AS column_name
FROM
    sys.tables AS t

    INNER JOIN sys.columns c 
        ON t.OBJECT_ID = c.OBJECT_ID

WHERE 
    c.name IN ('TRN_NO','REF_TRN_NO') 
    and t.name NOT IN('MST_PATTERNS_TOOL_LIFE','MST_CORES_TOOL_LIFE')

ORDER BY
    schema_name, table_name; 
    

DECLARE @min_no int,@max_no int   
SET @min_no = (select min(ID) from @T )            
SET @max_no = (select max(ID) from @T )   
PRINT  @min_no
PRINT  @max_no
while @min_no <= @max_no            
BEGIN

	DECLARE @FIND_COUNT INT
	SET @FIND_COUNT = 0
	DECLARE @TEST INT
	DECLARE @TABLE_NAME nvarchar(max)
	select @TABLE_NAME = TABLE_NAME from @t where id= @min_no
	DECLARE @TABLE_NAME_QRY nvarchar(max)

	--PRINT @TABLE_NAME
	SET @TABLE_NAME_QRY = 'SELECT TRN_NO FROM ' + @TABLE_NAME + ' WHERE TRN_NO = ' + @TRN_NO	
	DECLARE @TABLE_NAME_SHOW nvarchar(max)

	SET @TABLE_NAME_SHOW = 'SELECT * FROM ' + @TABLE_NAME + ' WHERE TRN_NO = ' + @TRN_NO
	INSERT INTO @T_TRN_NO
	EXEC SP_EXECUTESQL @TABLE_NAME_QRY    
	SET @FIND_COUNT = @@ROWCOUNT
		IF @FIND_COUNT > 0
		BEGIN
			INSERT INTO @T_FINAL
			SELECT @TABLE_NAME_SHOW, 'MASTER'
		END
		

	-- FOR REFERENCE START
	DECLARE @FIND_COUNT_REF INT
	SET @FIND_COUNT_REF = 0
	DECLARE @TEST_REF INT
	DECLARE @TABLE_NAME_QRY_REF nvarchar(max)
	SET @TABLE_NAME_QRY_REF = 'SELECT REF_TRN_NO FROM ' + @TABLE_NAME + ' WHERE REF_TRN_NO = ' + @TRN_NO	
	
	DECLARE @TABLE_NAME_SHOW_ref nvarchar(max)
	SET @TABLE_NAME_SHOW_ref = 'SELECT * FROM ' + @TABLE_NAME + ' WHERE REF_TRN_NO = ' + @TRN_NO
	
	 BEGIN TRY
	    INSERT INTO @T_TRN_NO
		EXEC SP_EXECUTESQL @TABLE_NAME_QRY_REF    
		SET @FIND_COUNT_REF = @@ROWCOUNT
		IF @FIND_COUNT_REF > 0
		BEGIN
			INSERT INTO @T_FINAL
			SELECT @TABLE_NAME_SHOW_ref, 'REFERENCE'
		END
	 END TRY
	 BEGIN CATCH
	 END CATCH
	 -- FOR REFERENCE END
	 	
	SET @min_no=@min_no+1
END

SELECT DISTINCT * FROM @T_FINAL
ORDER BY TABLE_TYPE

GO
