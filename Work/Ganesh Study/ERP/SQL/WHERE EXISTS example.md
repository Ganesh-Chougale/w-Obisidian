### `WHERE EXISTS` example
where exists syntax example  
```sql
DECLARE @LocationCode INT = '1010000024';
DECLARE @SalaryPeriod BigInt = '101242515710060001';

SELECT
	OTH.TRN_NO,
	OTH.EMPLOYEE_ID,
	OTH.MONTH_DATE
FROM
	TRN_HRM_OT_H AS OTH

	LEFT JOIN ERP_MASTERS..MST_EMPLOYEE as MST_EMP 
	ON OTH.EMPLOYEE_ID = MST_EMP.EMPLOYEE_ID
WHERE
	MST_EMP.LOCATION = @LocationCode
	AND OTH.STATUS_CODE >= '101'
	AND EXISTS(
				SELECT
					FROM_DATE,
					TO_DATE
				FROM
					TRN_SALARY_PERIOD
				WHERE
					TRN_NO=@SalaryPeriod
					AND OTH.MONTH_DATE >= TRN_SALARY_PERIOD.FROM_DATE
					AND OTH.MONTH_DATE <= TRN_SALARY_PERIOD.TO_DATE
			  );
```
