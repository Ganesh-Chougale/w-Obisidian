## 1. GET Index
```csharp
public async Task<ActionResult> Index(int? DepartmentCode, int? page)
{
    int DeptCode = 0;
    List<Department> _lstDepartment = Department.GetDepartmentList();

    if (DepartmentCode != null)
    {
        DeptCode = Convert.ToInt32(DepartmentCode);
    }
    else
    {
        if (_lstDepartment != null && _lstDepartment.Count > 0)
        {
            DeptCode = _lstDepartment.ElementAt(0).DepartmentId;
        }
    }

    List<DepartmentWiseMinMax> _lstDeptMinMax =
        await DepartmentWiseMinMax.GetDeptWsMinMaxLevel(DeptCode);

    int pageSize = 50;
    int pageNumber = 1;

    if (page != null)
    {
        pageNumber = Convert.ToInt16(page);
    }

    ViewData["DeptCode"] = DeptCode;
    ViewData["departments"] = _lstDepartment;

    return PartialView(_lstDeptMinMax.ToPagedList(pageNumber, pageSize));
}
```


```csharp
public async Task<ActionResult> Index(int? DepartmentCode, int? page, string searchString)
{
    int DeptCode = 0;
    List<Department> _lstDepartment = Department.GetDepartmentList();

    if (DepartmentCode != null)
    {
        DeptCode = Convert.ToInt32(DepartmentCode);
    }
    else
    {
        if (_lstDepartment != null && _lstDepartment.Count > 0)
        {
            DeptCode = _lstDepartment.ElementAt(0).DepartmentId;
        }
    }

    List<DepartmentWiseMinMax> _lstDeptMinMax =
        await DepartmentWiseMinMax.GetDeptWsMinMaxLevel(DeptCode);

    if (!String.IsNullOrEmpty(searchString))
    {
        _lstDeptMinMax = _lstDeptMinMax.Where(x => x.MaterialName.ToUpper().Contains(searchString.ToUpper())).ToList();
    }

    int pageSize = 50;
    int pageNumber = 1;

    if (page != null)
    {
        pageNumber = Convert.ToInt16(page);
    }

    ViewData["DeptCode"] = DeptCode;
    ViewData["departments"] = _lstDepartment;
    ViewBag.CurrentFilter = searchString;

    return PartialView(_lstDeptMinMax.ToPagedList(pageNumber, pageSize));
}
```

# 5. filterTable()

```javascript
function filterTable() {
    var searchString = $("#searchString").val();
    var deptcode = $("#ddlDepartment").val();

    var urlName = "DepartmentWiseMinMaxConfig/Index?DepartmentCode=" + deptcode + "&searchString=" + encodeURIComponent(searchString);

    $.ajax({
        url: urlName,
        type: "GET",
        success: function (response) {
            $("#mainContent").html(response);
        },
        error: function (err) {
            $("#mainContent").html(err);
        }
    });
}
```
