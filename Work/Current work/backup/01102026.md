                            STATUS_CODE = bObj.StatusCode
                        };
                        if (flag == 1)

                            _dalPdH.Create(dtoObj_H);
                        }
                        else if (flag == 2)
                        {
                            _dalPdI.DelPermenantTrnData(bObj.TrnNo, bObj.AmendNo);
                            _dalPdH.Edit(dtoObj_H);
                        }

                        if (bObj.LstPartPriceI != null)
                        {
                            DTOTrnPartPriceI _objI = new DTOTrnPartPriceI()
                            {
                                TRN_NO = item.TrnNo,
                                SR_NO = intISrNo,
