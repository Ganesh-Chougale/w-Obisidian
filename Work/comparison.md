CREATE PROCEDURE [dbo].[SP_GET_PACKING_DETAILS_OUT_SOURCE_PENDING_SCHEDULE] 
@MONTH_DATE      	   varchar(20),
@COMPANY_ID            int,
@SYSTEM_DB_NAME        VARCHAR(20)
AS
BEGIN
    
select 
  TEMP_PO.TRN_NO AS PO_NO ,
 TEMP_PO.TRN_AMEND_NO AS AMEND_NO,
 TEMP_PO.TRN_DATE AS PO_DATE ,
 CHILD_OUT_SOURCE.MATERIAL_CODE,
 SUM(CHILD_OUT_SOURCE.PLAN_QTY) AS SUM_OF_PLAN_QTY,
 CHILD_OUT_SOURCE.SUB_GL_ACNO,
 SCH_H.TRN_NO AS SCH_NO ,
 SCH_H.AMEND_NO AS SCH_AMEND_NO,
 SCH_H.BOOKED_SCH_QTY
 FROM
	TRN_PRD_CHILD_PART_OUT_SOURCE AS CHILD_OUT_SOURCE 
	LEFT OUTER JOIN(
					SELECT
						POH.TRN_NO,
						POH.TRN_AMEND_NO,
						POH.TRN_DATE,
						POH.SUB_GL_ACNO,
						MAT_I.MATERIAL_CODE
					FROM
						TRN_PO_H AS POH
						INNER JOIN TRN_PO_MAT_I AS MAT_I ON MAT_I.TRN_NO = POH.TRN_NO
						AND MAT_I.AMEND_NO = POH.TRN_AMEND_NO
					WHERE 
						SUBSTRING(CONVERT(VARCHAR,POH.TRN_NO),8,3) = '701'
						AND POH.STATUS_CODE IN (0)
						AND LEFT(MAT_I.MATERIAL_CODE,5) = '10101'
						
	                ) AS TEMP_PO 
		ON TEMP_PO.SUB_GL_ACNO = CHILD_OUT_SOURCE.SUB_GL_ACNO 
	    AND TEMP_PO.MATERIAL_CODE = CHILD_OUT_SOURCE.MATERIAL_CODE
		LEFT JOIN (
			SELECT DISTINCT 
				SCHEDULE_H.TRN_NO,
				SCHEDULE_H.AMEND_NO,
				SCHEDULE_H.TRN_SUB_TYPE,
				SCHEDULE_H.REF_TRN_NO,
				SCHEDULE_H.REF_AMEND_NO,
				TRN_SCHEDULE_I.MATERIAL_CODE,
				SUM(TRN_SCHEDULE_I.SCH_QTY) AS BOOKED_SCH_QTY
			FROM
				TRN_SCHEDULE_H AS SCHEDULE_H
				INNER JOIN TRN_SCHEDULE_I ON SCHEDULE_H.TRN_NO=TRN_SCHEDULE_I.TRN_NO
					AND SCHEDULE_H.AMEND_NO=TRN_SCHEDULE_I.AMEND_NO
			WHERE
				LEFT(TRN_SCHEDULE_I.SCH_DATE,6) = LEFT(@MONTH_DATE,6)
				AND SUBSTRING(CONVERT(VARCHAR,TRN_SCHEDULE_I.TRN_NO),8,3) = '903'	
				AND TRN_SCHEDULE_I.TRN_SUB_TYPE IN (14,17)
				AND SCHEDULE_H.STATUS_CODE IN (0) 
				AND TRN_SCHEDULE_I.STATUS_CODE IN (0) 
				GROUP BY
						SCHEDULE_H.TRN_NO,
						SCHEDULE_H.TRN_SUB_TYPE,
						SCHEDULE_H.REF_TRN_NO,
						SCHEDULE_H.AMEND_NO,
						SCHEDULE_H.REF_AMEND_NO,
						TRN_SCHEDULE_I.MATERIAL_CODE
		) SCH_H
		 ON SCH_H.REF_TRN_NO = TEMP_PO.TRN_NO 
		AND SCH_H.REF_AMEND_NO = TEMP_PO.TRN_AMEND_NO
		AND SCH_H.MATERIAL_CODE = CHILD_OUT_SOURCE.MATERIAL_CODE
 WHERE 
    CHILD_OUT_SOURCE.STATUS_CODE IN (0)
	AND LEFT (CHILD_OUT_SOURCE.PLAN_DATE,6) = LEFT(@MONTH_DATE, 6)	
 GROUP BY 
	TEMP_PO.TRN_NO,
    TEMP_PO.TRN_AMEND_NO,
	TEMP_PO.TRN_DATE,
    CHILD_OUT_SOURCE.MATERIAL_CODE,	
	CHILD_OUT_SOURCE.SUB_GL_ACNO,
	SCH_H.TRN_NO,
    SCH_H.AMEND_NO,
	SCH_H.BOOKED_SCH_QTY
ORDER BY 
    CHILD_OUT_SOURCE.MATERIAL_CODE

	
END






GO
