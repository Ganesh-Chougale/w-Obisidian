public static async Task<List<PackingCost>> CastingWeightVsActualWeightList(string CustomerIds)
{
    try
    {
        ICastingWeightChangeHistory _dalCastingWeightChangeHistory = new DALCastingWeightChangeHistory();
        List<DTOCastingWeightChangeHistory> lstDtoData = await Task.Run(() => { return _dalCastingWeightChangeHistory.CastingWeightVsActualWeightList(CustomerIds); });

        List<PackingCost> lstCastingWeightChangeHistory = new List<PackingCost>();

        if (lstDtoData.Count > 0)
        {
            lstCastingWeightChangeHistory = (from objDto in lstDtoData
                                                select new PackingCost
                                                {
                                                    SubGlAcno = objDto.SUB_GL_ACNO,
                                                    CustomerName = objDto.CUSTOMER_NAME,
                                                    MaterialCode = objDto.MATERIAL_CODE,
                                                    MaterialName = objDto.MATERIAL_NAME,
                                                    SrNo = objDto.SR_NO,
                                                    TrnDate = objDto.TRN_DATE,
                                                    FormattedDate = DateUtility.getFormatedDate(objDto.TRN_DATE, 0),
                                                    CastingWeightTime = objDto.CASTING_WEIGHT_TIME,                                                         
                                                    CastingWeight = objDto.CASTING_WEIGHT,
                                                    ProjectCastingWeight = objDto.PROJECT_CASTING_WEIGHT,
                                                    ApprovedCastingWeight = objDto.APPROVED_CASTING_WEIGHT,
                                                    SaleQty = objDto.SALE_QTY,
                                                    Rate = objDto.RATE
                                                }).ToList();
            return lstCastingWeightChangeHistory;
        }
        else
        {
            return lstCastingWeightChangeHistory;
        }


    }
    catch (Exception ex)
    {
        throw new Exception("Request Failed : " + ex);
    }
}
