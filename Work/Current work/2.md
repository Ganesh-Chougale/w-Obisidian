```sql
SELECT 
    DISTINCT TOP 1 MST_MATERIALS.MATERIAL_CODE
    CASE 
        WHEN
            TRN_MAT_I_PURCHASE.RATE_PER <> 0 
        THEN 
        ISNULL(CONVERT(DECIMAL(20,2),TRN_MAT_I_PURCHASE.RATE /  TRN_MAT_I_PURCHASE.RATE_PER),0)
    ELSE 
        0 
    END AS RATE
FROM 
    ERP_MASTERS.DBO.MST_MATERIALS  MST_MATERIALS

LEFT OUTER JOIN (
                SELECT
                    TRN_MAT_I_PURCHASE.MATERIAL_CODE ,
                    TRN_MAT_I_PURCHASE.TRN_DATE,
                    MAX(TRN_NO) AS TRN_NO                    
                FROM  
                    TRN_MAT_I_PURCHASE TRN_MAT_I_PURCHASE

                    INNER JOIN (
                                SELECT
                                    TRN_MAT_I_PURCHASE.MATERIAL_CODE,
                                    MAX(TRN_DATE)  TRN_DATE
                                FROM
                                    TRN_MAT_I_PURCHASE  TRN_MAT_I_PURCHASE 
                                WHERE 
                                    STATUS_CODE = 0  
                                    AND SUBSTRING(CONVERT(VARCHAR,TRN_MAT_I_PURCHASE.TRN_NO),8,3)  = 206 
                                    AND TRN_MAT_I_PURCHASE.TRN_SUB_TYPE IN (" + strTranSubTypes + ") AND TRN_MAT_I_PURCHASE.MATERIAL_CODE= @MatCode  
                                GROUP BY
                                    TRN_MAT_I_PURCHASE.MATERIAL_CODE 
                                ) TMP 
                    ON  TRN_MAT_I_PURCHASE.MATERIAL_CODE= TMP.MATERIAL_CODE
                        AND TMP.TRN_DATE = TRN_MAT_I_PURCHASE.TRN_DATE
                WHERE
                    STATUS_CODE = 0 
                    AND TRN_SUB_TYPE IN (" + strTranSubTypes + ")
                GROUP BY
                    TRN_MAT_I_PURCHASE.MATERIAL_CODE,
                    TRN_MAT_I_PURCHASE.TRN_DATE
                ) TMPRATE   
ON  MST_MATERIALS.MATERIAL_CODE= TMPRATE.MATERIAL_CODE 

LEFT OUTER JOIN 
    TRN_MAT_I_PURCHASE TRN_MAT_I_PURCHASE  
    ON TRN_MAT_I_PURCHASE.MATERIAL_CODE= TMPRATE.MATERIAL_CODE
    AND TMPRATE.TRN_NO = TRN_MAT_I_PURCHASE.TRN_NO                    
    AND TMPRATE.TRN_DATE = TRN_MAT_I_PURCHASE.TRN_DATE 
WHERE
    MST_MATERIALS.MATERIAL_CODE=  @MatCode
```   
