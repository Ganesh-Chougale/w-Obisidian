CREATE PROCEDURE [dbo].[SP_PACKING_DETAILS_OUT_SOURCE_PURCHASE_SCHEDULE_TAB_DATA] 
@MONTH_DATE      	   varchar(20),
@COMPANY_ID            int,
@SYSTEM_DB_NAME        VARCHAR(20)
AS
BEGIN
    
SELECT 
    TEMP_PO.TRN_NO AS PO_NO ,
    TEMP_PO.TRN_AMEND_NO AS AMEND_NO,
    TEMP_PO.TRN_DATE AS PO_DATE ,
    SAVED_TBL.BOM_MATERIAL_CODE,
    SUM(SAVED_TBL.PLAN_QTY) AS SUM_OF_PLAN_QTY,
    SAVED_TBL.SUPPLIER_SUB_GL_ACNO,
    SCH_H.TRN_NO AS SCH_NO ,
    SCH_H.AMEND_NO AS SCH_AMEND_NO,
    SCH_H.SCH_DATE,
    SCH_H.SCH_QTY
FROM
	TRN_PRD_PACKING_MAT_OUT_SOURCE AS SAVED_TBL 

	INNER JOIN(
                SELECT
                    POH.TRN_NO,
                    POH.TRN_AMEND_NO,
                    POH.TRN_DATE,
                    POH.SUB_GL_ACNO,
                    MAT_I.MATERIAL_CODE
                FROM
                    TRN_PO_H AS POH

                    INNER JOIN TRN_PO_MAT_I AS MAT_I 
                    ON MAT_I.TRN_NO = POH.TRN_NO
                    AND MAT_I.AMEND_NO = POH.TRN_AMEND_NO

                WHERE 
                    SUBSTRING(CONVERT(VARCHAR,POH.TRN_NO),8,3) = '701'
                    AND POH.STATUS_CODE IN (0)
                    AND LEFT(MAT_I.MATERIAL_CODE,5) = '10103'
    ) AS TEMP_PO 
    ON TEMP_PO.SUB_GL_ACNO = SAVED_TBL.SUPPLIER_SUB_GL_ACNO 
    AND TEMP_PO.MATERIAL_CODE = SAVED_TBL.BOM_MATERIAL_CODE

    INNER JOIN (
                SELECT 
                    SCH_H.TRN_NO,
                    SCH_H.AMEND_NO,
                    SCH_H.TRN_SUB_TYPE,
                    SCH_H.REF_TRN_NO,
                    SCH_H.REF_AMEND_NO,
                    SCH_I.MATERIAL_CODE,
                    SCH_I.SCH_DATE,
                    SCH_I.SCH_QTY
                FROM
                    TRN_SCHEDULE_H AS SCH_H

                    INNER JOIN TRN_SCHEDULE_I AS SCH_I
                    ON SCH_H.TRN_NO=SCH_I.TRN_NO
                    AND SCH_H.AMEND_NO=SCH_I.AMEND_NO

                WHERE
                    LEFT(SCH_I.SCH_DATE,6) = @MONTH_DATE
                    AND SUBSTRING(CONVERT(VARCHAR,SCH_I.TRN_NO),8,3) = '903'	
                    AND SCH_I.TRN_SUB_TYPE IN (14,17)
                    AND SCH_H.STATUS_CODE IN (0) 
                    AND SCH_I.STATUS_CODE IN (0) 
    ) SCH_H 
    ON SCH_H.REF_TRN_NO = TEMP_PO.TRN_NO 
    AND SCH_H.REF_AMEND_NO = TEMP_PO.TRN_AMEND_NO
    AND SCH_H.MATERIAL_CODE = SAVED_TBL.BOM_MATERIAL_CODE

 WHERE 
    SAVED_TBL.STATUS_CODE IN (0)
	AND LEFT (SAVED_TBL.PLAN_DATE,6) = @MONTH_DATE
 GROUP BY 
	TEMP_PO.TRN_NO,
    TEMP_PO.TRN_AMEND_NO,
	TEMP_PO.TRN_DATE,
    SAVED_TBL.BOM_MATERIAL_CODE,	
	SAVED_TBL.SUPPLIER_SUB_GL_ACNO,
	SCH_H.TRN_NO,
    SCH_H.AMEND_NO,
	SCH_H.SCH_DATE,
	SCH_H.SCH_QTY 	
ORDER BY 
    SAVED_TBL.BOM_MATERIAL_CODE

	
END





GO
