```csharp
    function calculateTotalPackingCost() {

        var totalPackingCost = 0;
        var totalPartQuantity = 0;

        var rowCount = $("#PackingMaterialBomDetails tbody tr").length;

        $("#PackingMaterialBomDetails tbody tr").each(function () {

            var cost = parseFloat($(this).find(".costPerPiece").text()) || 0;

            totalPackingCost += cost;
        });

        $("#PartQuantityTable tbody tr").each(function () {

            var partQuantity = parseFloat($(this).find(".partQuantityVal").val()) || 0;

            totalPartQuantity += partQuantity;
        });

        $("#totalPackingCost").text(totalPackingCost.toFixed(2));

        var costPerPiece = 0;

        if (totalPartQuantity > 0) {
            costPerPiece = totalPackingCost / totalPartQuantity;
        }

        $("#costPerPiece").text(costPerPiece.toFixed(2));

        if (rowCount > 0) {

            $("#FooterLine").css("display", "block");

        } else {

            $("#FooterLine").hide();

            $("#totalPackingCost").text("0.00");
            $("#costPerPiece").text("0.00");
        }
    }
```  
